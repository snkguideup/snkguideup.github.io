rules_version = '2';

service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function myRole() {
      return get(/databases/$(database)/documents/users/$(request.auth.uid)).data.role;
    }

    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && myRole() in ['admin', 'superadmin'];
    }

    function isSuper() {
      return signedIn()
        && exists(/databases/$(database)/documents/users/$(request.auth.uid))
        && myRole() == 'superadmin';
    }

    match /users/{uid} {
      allow read: if signedIn() && (request.auth.uid == uid || isAdmin());
      allow create: if signedIn() && request.auth.uid == uid
                    && request.resource.data.role == 'user';
      allow update: if signedIn() && (
                      (request.auth.uid == uid && request.resource.data.role == resource.data.role)
                      || isSuper()
                    );
      allow delete: if isSuper();
    }

    match /site/{doc}         { allow read: if true; allow write: if isAdmin(); }
    match /courses/{doc}      { allow read: if true; allow write: if isAdmin(); }
    match /testimonials/{doc} { allow read: if true; allow write: if isAdmin(); }
    match /partners/{doc}     { allow read: if true; allow write: if isAdmin(); }
    match /blogs/{doc}        { allow read: if true; allow write: if isAdmin() || (signedIn() && exists(/databases/$(database)/documents/users/$(request.auth.uid)) && myRole() == 'mentor'); }

    match /{document=**} {
      allow read, write: if false;
    }
  }
}
